  <meta name="viewport" content="width=device-width, initial-scale=1.0, maximum-scale=1.0, user-scalable=no">

  <title>Stickman Cactus & Knife Runner</title>

  <style>

    body {

      margin: 0;

      text-align: center;

      background: #fff;

      touch-action: none;

      -webkit-user-select: none;

      user-select: none;

      font-family: sans-serif;

    }

    canvas {

      border: 2px solid #000;

      background: #fff;

      margin-top: 10px;

      max-width: 95%;

      height: auto;

    }

  </style>

<body>

  <canvas id="gameCanvas" width="480" height="200"></canvas>

  <script>

    const canvas = document.getElementById('gameCanvas');

    const ctx = canvas.getContext('2d');


    // プレイヤー（棒人間）状態

    let player = {

      x: 60,

      y: 160,

      dy: 0,

      gravity: 0.5,

      firstJumpPower: -9,

      secondJumpPower: -10,

      isGrounded: false,

      jumpCount: 0,

      maxJumps: 2,

      runFrame: 0

    };


    let obstacles = [];

    let frame = 0;

    let score = 0;

    let highScore = localStorage.getItem('stickman_cactus_highscore') || 0;

    let gameOver = false;


    // 棒人間描画

    function drawPlayer() {

      ctx.strokeStyle = '#000';

      ctx.lineWidth = 2;

      ctx.lineCap = 'round';

      ctx.lineJoin = 'round';


      let headX = player.x;

      let headY = player.y - 30;


      // 頭

      ctx.beginPath();

      ctx.arc(headX, headY, 7, 0, Math.PI * 2);

      ctx.stroke();


      // 体

      ctx.beginPath();

      ctx.moveTo(headX, headY + 7);

      ctx.lineTo(headX, player.y - 8);

      ctx.stroke();


      if (player.isGrounded) {

        player.runFrame += 0.25;

        let legAngle = Math.sin(player.runFrame) * 12;

        let armAngle = Math.sin(player.runFrame) * 10;


        // 脚

        ctx.beginPath();

        ctx.moveTo(headX, player.y - 8);

        ctx.lineTo(headX - legAngle, player.y);

        ctx.moveTo(headX, player.y - 8);

        ctx.lineTo(headX + legAngle, player.y);

        ctx.stroke();


        // 腕

        ctx.beginPath();

        ctx.moveTo(headX, headY + 12);

        ctx.lineTo(headX + armAngle, headY + 20);

        ctx.moveTo(headX, headY + 12);

        ctx.lineTo(headX - armAngle, headY + 20);

        ctx.stroke();

      } else {

        // ジャンプポーズ

        ctx.beginPath();

        ctx.moveTo(headX, player.y - 8);

        ctx.lineTo(headX - 6, player.y - 2);

        ctx.lineTo(headX - 10, player.y + 4);


        ctx.moveTo(headX, player.y - 8);

        ctx.lineTo(headX + 6, player.y - 4);

        ctx.lineTo(headX + 8, player.y + 2);

        ctx.stroke();


        ctx.beginPath();

        ctx.moveTo(headX, headY + 12);

        ctx.lineTo(headX - 10, headY + 5);

        ctx.moveTo(headX, headY + 12);

        ctx.lineTo(headX + 10, headY + 5);

        ctx.stroke();

      }

    }


    // サボテン描画

    function drawCactus(obs) {

      ctx.fillStyle = '#000';

      ctx.strokeStyle = '#000';

      ctx.lineWidth = 2;


      ctx.fillRect(obs.x + obs.w * 0.35, obs.y, obs.w * 0.3, obs.h);


      if (obs.h >= 25) {

        ctx.fillRect(obs.x, obs.y + obs.h * 0.3, obs.w * 0.4, obs.h * 0.12);

        ctx.fillRect(obs.x, obs.y + obs.h * 0.15, obs.w * 0.12, obs.h * 0.25);


        ctx.fillRect(obs.x + obs.w * 0.6, obs.y + obs.h * 0.45, obs.w * 0.4, obs.h * 0.12);

        ctx.fillRect(obs.x + obs.w * 0.88, obs.y + obs.h * 0.25, obs.w * 0.12, obs.h * 0.25);

      }

    }


    // 飛んでくるナイフ描画関数

    function drawKnife(obs) {

      ctx.fillStyle = '#000';

      ctx.strokeStyle = '#000';

      ctx.lineWidth = 1.5;


      let x = obs.x;

      let y = obs.y;


      // 刃（左向きの鋭い形状）

      ctx.beginPath();

      ctx.moveTo(x, y + 6);                  // 刃先（一番左）

      ctx.lineTo(x + 18, y + 1);             // 刃の背

      ctx.lineTo(x + 18, y + 11);            // 刃の腹（鍔）

      ctx.closePath();

      ctx.fill();


      // 持ち手（柄）

      ctx.fillRect(x + 18, y + 4, 8, 4);

      // 柄頭

      ctx.fillRect(x + 26, y + 3, 2, 6);

    }


    function update() {

      if (gameOver) return;

      frame++;


      player.dy += player.gravity;

      player.y += player.dy;


      if (player.y >= 160) {

        player.y = 160;

        player.dy = 0;

        player.isGrounded = true;

        player.jumpCount = 0;

      }


      let gameSpeed = 5 + Math.floor(score / 5) * 0.6;

      let spawnInterval = Math.max(50, 95 - Math.floor(score / 4) * 4);


      if (frame % spawnInterval === 0) {

        let isKnife = score >= 5 && Math.random() < 0.35;


        if (isKnife) {

          // ナイフ（高度2パターン：ジャンプで避ける / しゃがむかそのまま通過）

          let knifeY = Math.random() < 0.5 ? 90 : 115;

          obstacles.push({ x: canvas.width, y: knifeY, w: 28, h: 12, type: 'knife' });

        } else {

          // サボテン

          let cactusType = Math.random();

          let h, w;

          if (cactusType < 0.3) {

            h = 45; w = 24;

          } else if (cactusType < 0.7) {

            h = 32; w = 20;

          } else {

            h = 22; w = 16;

          }

          obstacles.push({ x: canvas.width, y: 160 - h, w: w, h: h, type: 'cactus' });

        }

      }


      for (let i = obstacles.length - 1; i >= 0; i--) {

        obstacles[i].x -= gameSpeed;


        let pLeft = player.x - 6;

        let pRight = player.x + 6;

        let pTop = player.y - 35;

        let pBottom = player.y;


        let oLeft = obstacles[i].x;

        let oRight = obstacles[i].x + obstacles[i].w;

        let oTop = obstacles[i].y;

        let oBottom = obstacles[i].y + obstacles[i].h;


        if (pRight > oLeft && pLeft < oRight && pBottom > oTop && pTop < oBottom) {

          gameOver = true;

          if (score > highScore) {

            highScore = score;

            localStorage.setItem('stickman_cactus_highscore', highScore);

          }

        }


        if (obstacles[i].x < -30) {

          obstacles.splice(i, 1);

          score++;

        }

      }

    }


    function draw() {

      ctx.clearRect(0, 0, canvas.width, canvas.height);


      // 地面

      ctx.strokeStyle = '#000';

      ctx.lineWidth = 2;

      ctx.beginPath();

      ctx.moveTo(0, 160);

      ctx.lineTo(canvas.width, 160);

      ctx.stroke();


      drawPlayer();


      obstacles.forEach(obs => {

        if (obs.type === 'knife') {

          drawKnife(obs);

        } else {

          drawCactus(obs);

        }

      });


      // UI

      ctx.fillStyle = '#000';

      ctx.font = "bold 14px sans-serif";

      ctx.fillText("SCORE: " + score, 10, 25);

      ctx.fillText("HI-SCORE: " + highScore, 130, 25);


      let remainingJumps = player.maxJumps - player.jumpCount;

      ctx.font = "12px sans-serif";

      ctx.fillText("JUMP: " + "★".repeat(remainingJumps), 380, 25);


      if (gameOver) {

        ctx.fillStyle = 'rgba(255, 255, 255, 0.85)';

        ctx.fillRect(0, 0, canvas.width, canvas.height);


        ctx.fillStyle = '#000';

        ctx.font = "bold 22px sans-serif";

        ctx.fillText("GAME OVER", 170, 80);


        ctx.font = "15px sans-serif";

        ctx.fillText("Score: " + score + "  (Best: " + highScore + ")", 160, 110);

        ctx.fillText("Tap to Restart", 185, 145);

      }

    }


    function loop() {

      update();

      draw();

      requestAnimationFrame(loop);

    }


    function handleAction(e) {

      if (e) e.preventDefault();


      if (gameOver) {

        obstacles = [];

        score = 0;

        frame = 0;

        player.y = 160;

        player.dy = 0;

        player.isGrounded = true;

        player.jumpCount = 0;

        gameOver = false;

      } else {

        if (player.jumpCount < player.maxJumps) {

          player.dy = (player.jumpCount === 0) ? player.firstJumpPower : player.secondJumpPower;

          player.isGrounded = false;

          player.jumpCount++;

        }

      }

    }


    canvas.addEventListener('pointerdown', handleAction, { passive: false });

    window.addEventListener('keydown', (e) => { if (e.code === 'Space') handleAction(e); });


    loop();

  </script>
